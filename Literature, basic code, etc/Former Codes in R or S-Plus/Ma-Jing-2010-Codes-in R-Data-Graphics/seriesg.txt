bopt=result1$bopt  ## The selected bandwidth

#### Fit an ARMA model to the residuals with given p and q
p=1
q=1
ARMA.res11=arima(Res, order=c(p, 0, q))

summary(ARMA.res11)


#### Calculate BIC for ARMA(p, q), p=0, 1, ..., p.max; q=0, ..., q.max
p.max=5
q.max=1

BIC=matrix(0, p.max+1, q.max+1)

for(p in 0:p.max)
  {for(q in 0:q.max)
    {ARMA.respq=arima(Res, order=c(p, 0, q))
     BIC[p+1, q+1]=-2*ARMA.respq$loglik+log(n)*(p+q)
    }
   }


BIC
phat=2
qhat=0
ARMA.res.BIC=arima(Res, order=c(phat, 0, qhat))
ARMA.res.BIC


ARMA.air=arima(x = Res, order = c(phat, 0, qhat))
ARMA.air$coef




ged=ge[n]-ge[n-1]
gef=1:s

for(i in 1:s)
{gef[i]=ge[n]+i*ged}
gef

sef=se[(n-s+1):n]
sef

Res

ref=1:s

mue0=ARMA.air$coef[phat+1]
ar1=ARMA.air$coef[1]
ar2=ARMA.air$coef[2]

ref[1]=mue0+ar1*(Res[n]-mue0)+ar2*(Res[n-1]-mue0)
ref[2]=mue0+ar1*(ref[1]-mue0)+ar2*(Res[n]-mue0)

for(i in 3:s)
{ref[i]=mue0+ar1*(ref[i-1]-mue0)+ar2*(ref[i-2]-mue0)}
ref

yef=gef+sef+ref


gs=ge+se
gs=c(gs,NA*(1:s))
yef=c(NA*(1:n),yef)
yef
