############## Begin of the function DeSeaTS(y,s,p,kn,h) ##############
DeSeaTS_function(y,s,p,kn,h)
{
######################### The inputs  #################################
## y - the input seasonal time series data
## s - the seasonal period
## p - 1 or 3, degree of polynomial for estimating the trend function.
## kn- 1, 2, 3 or 4, number of the kernel used as weighting function
         ## kn=1: Uniform Kernel       kn=2: Epanecknikov Kernel
         ## kn=3: Bisquare Kernel      kn=4: Triweight Kernel
## h - the bandwidth
         ## Tip 1: h should be smaller than n/2
         ## Tip 2: Try a bandwidth between 1.5*s and 6*s
 
######################## The basic ideas  ###########################
#### The model: yi = gi + Si +ei                                 ####
#### The approach used here is LWR (locally weighted regression) ####
#### LWR is a linear smoother. The main tesk is to calculate the ####
#### weighting systems for estimating g and S at each point t.   ####
#### In the following only the weighting systems at the left     ####
#### boundary (i=1 to h) and that at i=h+1 will be computed.     ####
#### The weighting system at a point h+1=<i<=n-h is the same.    ####
#### The weightings at i>n-hh are symmetric to that at j=n-i+1.  #### 
#### See Feng (1999), Chapter 7.6 + Feng & Heiler (2000), 786-8. ####

################## constants, vectors and matrices ##################
n_length(y)           ## number of observations
ye_matrix(0,2,n)      ## definition of the matrix of estimates
ht_2*h+1              ## total bandwith

xg_matrix(0,p+s,ht)   ## data matrix for the trend
xs_xg                 ## data matrix for the season
xw_xg                 ## weighted data matrix


  for(j in (1:(p+1))){xg[j,1:ht]_(((1:ht)-h-1)/h)**(j-1)}
  for(j in (1:(trunc((s-1)/2)))){
	xs[p+2*j,  1:ht]_cos(2*j*pi*(1:ht)/s)
	xs[p+2*j+1,1:ht]_sin(2*j*pi*(1:ht)/s)
	}

  if(s/2==trunc(s/2)){
	xs[p+s,1:ht]_cos(pi*(1:ht))
	}

xt_rbind(xg[1:(p+1),], xs[(p+2):(p+s),])



hr_c(2*h-(0:h)) #pointwise right bandwith

     for(i in (1:(n-h-1))){
	     if(i<=h+1){
                   wk_(1-(((1:ht)-i)/(hr[i]+0.5))**2)**(kn-1)

                   for(j in (1:(p+s))){xw[j,]_xt[j,]*wk}



                   yw_solve(xw%*%t(xt))%*%xw
       
                   xgt_t(xg[,i])
                   xst_t(xs[,i])

                   gw_xgt%*%yw
                   ye[1,i]_gw%*%cbind(y[1:ht])
                   sw_xst%*%yw
                   ye[2,i]_sw%*%cbind(y[1:ht])

                   ye[1,n-i+1]_gw%*%cbind(y[n:(n-ht+1)])
                   ye[2,n-i+1]_sw%*%cbind(y[n:(n-ht+1)])

                  }

	     if(i>h+1 && i<n-h){
                           ye[1,i]_gw%*%cbind(y[(i-h):(i+h)])
                           ye[2,i]_sw%*%cbind(y[(i-h):(i+h)])
                          }
        }

#### final results
     result_ye

drop(result)
};
################ End of the function DeSeaTS(y,s,p,kn,h) ################


 
########### Begin of the function DerivativeG(y,s,v,pv,kn,hv) ###########
DerivativeG_function(y,s,v,pv,kn,hv)
{
######################### The inputs  #################################
## v, the order of the derivative
## y, s, pv, kn, hv are as before.

#### Here pv should be larger than v with pv-v odd.               ####
#### See Feng (1999), Chapter 4 + Heiler & Feng (1996, 2000).     ####

################## constants, vectors and matrices ##################
h_hv
p_pv

n_length(y)           ## number of observations
gv_matrix(0,1,n)      ## definition of estimates vector
ht_2*h+1              ## total bandwith

xg_matrix(0,p+s,ht)   ## data matrix for the trend
xs_xg                 ## data matrix for the season
xv_xg                 ## data matrix for the trend
xw_xg                 ## weighted data matrix



        th_(1:ht)-h-1
        for(j in (1:(p+1)))
            {
 	          xg[j,1:ht]_(th/n)**(j-1)
             if(j>v)
                  {
	              # Cj_factorial(j)/factorial(j-v)  # Originalversion
	                Cj_factorial(j-1)/factorial(j-1-v)  # Anpassung
     	             xv[j,1:ht]_Cj*(th/n)**(j-1-v)
                  }
	         }

        for(j in (1:(trunc((s-1)/2)))){
	       xs[p+2*j,  1:ht]_cos(2*j*pi*th/s)
	       xs[p+2*j+1,1:ht]_sin(2*j*pi*th/s)
	       }

        if(s/2==trunc(s/2)){
       	xs[p+s,1:ht]_cos(pi*th)
	       }

        xt_rbind(xg[1:(p+1),], xs[(p+2):(p+s),])


hr_c(2*h-(0:h)) #pointwise right bandwith

     for(i in (1:(n-h-1))){
	     if(i<=h+1){
                wk_(1-(((1:ht)-i)/(hr[i]+0.5))**2)**(kn-1)

                for(j in (1:(p+s))){xw[j,]_xt[j,]*wk}

                   yw_solve(xw%*%t(xt))%*%xw
                   xvt_t(xv[,i])
       
                   vw_xvt%*%yw
                   gv[i]_vw%*%cbind(y[1:ht])
                   gv[n-i+1]_vw%*%cbind(y[n:(n-ht+1)])*(-1)**v
                  }

	     if(i>h+1 && i<n-h){
                           gv[i]_vw%*%cbind(y[(i-h):(i+h)])
                          }
        }

     gvt_gv          #### Derivative w.r.t. t=i/n
     gvi_gv*(n)**(v) #### Derivative w.r.t. i

#### final results
     result_list(gvt=gvt, gvi=gvi)

drop(result)
};
############ End of the function DerivativeG(y,s,v,pv,kn,hv) ###########



############ Begin of the function VarSeaTSD(y,s) ###########
VarSeaTSD_function(y,s)
{
##################### The inputs  ############################
## y, s are as before.

#### Here two difference-based variance estimates are calculated. ####
#### See Feng (1999), Chapter 7 + Heiler & Feng (1996, 2000).     ####

n_length(y)
n1_s+3
n2_2*s+3

D1_(1:n1)*0
D2_(1:n2)*0

D1[1:3]_c(-1, 2, -1)
D1[(s+1):n1]_c(1, -2, 1)



D2[1:3]_c(-1, 2, -1)
D2[(s+1):(s+3)]_c(2, -4, 2)
D2[(2*s+1):n2]_c(-1, 2, -1)

D1_D1/sqrt(sum(D1**2))
D2_D2/sqrt(sum(D2**2))

sig2D1_0
sig2D2_0

for(i in 1:(n-n1+1)){sig2D1_sig2D1+(D1%*%cbind(y[i:(i+n1-1)]))**2}
for(i in 1:(n-n2+1)){sig2D2_sig2D2+(D2%*%cbind(y[i:(i+n2-1)]))**2}
	
sig2D1_sig2D1/(n-n1+1)
sig2D2_sig2D2/(n-n2+1)

#### final results
     result_list(sig2D1=sig2D1, sig2D2=sig2D2)

drop(result)
};
############ End of the function  VarSeaTSD(y,s) ###########


############ Begin of the function AutoDeSeaTS(y,s,p,kn,bStart) ###########
AutoDeSeaTS_function(y,s,p,kn,bStart)
       {
        ## y, s, p, kn are as before

        ##### input parameters
           bopt_bStart
           hvold_0
           n_length(y)
           k_p+1
           bmax_0.5-1/n     #### The maximal bandwidth 
           bmin_1*s/n       #### The minimal bandwidth 

           v_k
           pv_k+1
           runc_1
############# Delta >=0 
           if(k==2){d_0}   ## A positive number to
           if(k==4){d_0}   ## avoid the boundary effect
           ## 0<= d <=0.15, parameter for the length of the boundary
           n1_trunc(n*d)
           cat("n1=Delta*n=", n1, fill=T)
 

        ###### Kernel constants
           wk_0
           m_100000
           u_(-m:m)/(m+0.5)
           wk_(1-u^2)^(kn-1)
           wk_wk/sum(wk)*m
           R2_sum(wk^2)/(m-0.5)
   
           if(k==2){wkp_wk}
           if(k==4 && kn==2){
	          wkp_15/32*(3-10*u^2+7*u^4)
             wkp_wkp/sum(wkp)*m
            }
           if(k==4 && kn==3){
	          wkp_105/64*(1-5*u^2+7*u^4-3*u**6)
             wkp_wkp/sum(wkp)*m
            }
	  
           Rp_sum(wkp^2)/(m-0.5)
         # mukp_sum((u^4)*wkp)/m  # Originalversion
           mukp_sum((u^k)*wkp)/m  # Anpassung

        #### two constant in the bandwidth
           c1_factorial(k)**2/(2*k)
           c2_(1-2*d)*(Rp+(s-1)*R2)/(mukp)**2

        ####  difference-based estimation of the variance
                sigDC_1   #### 1 or 2
                if(sigDC==1){sig2D_VarSeaTSD(y,s)$sig2D1}
                if(sigDC==2){sig2D_VarSeaTSD(y,s)$sig2D2}
                sig2_sig2D

        #### the main iteration
           noi_20          ### maximal nuber of iterations

           for(i in 1:noi)
             {
              if(runc==1)
                { 	
                cat("This is the ", i, "th iteration.", fill=T)
         	       if(i>=2){hvold_hv}

########## Estimate the variance

############# The inflation method    ###############	
	             if(k==2){bv_1*bopt^(5/7)}   #### the most important idea
	             if(k==4){bv_bopt^(9/13)}  #### the most important idea

	              bv_max(bv, bmin)
                 bv_min(bv, bmax)
                 hv_trunc(bv*n+0.5)
                 bvP_trunc(bv*1000+0.5)/1000
                 cat("hv=", hv, "    bv=", bvP)

     
                 gv_DerivativeG(y, s, v, pv, kn, hv)$gvt
                 I2_sum(gv[n1:(n-n1)]**2)/n
                 

                 ### the optimal bandwidth
                    c3_sig2/I2 

                    bopt_(c1*c2*c3)**(1/(2*k+1))*n**(-1/(2*k+1))
                    bopt_max(bopt, bmin)
                    bopt_min(bopt, bmax)

                 I0_i

                 hopt_trunc(bopt*n+0.5)
                 boP_trunc(bopt*1000+0.5)/1000              
                 cat("    hopt=", hopt, "    bopt=", boP, fill=T)

                 dh_hvold-hv
                 if(i>=3 && dh==0){runc_0}
                }
             }

         #### smooth with the selected bandwidth
            hopt_trunc(bopt*n+0.5)

            ye_DeSeaTS(y, s, p, kn, hopt)

         #### final results
           result_list(ye=ye, bopt=bopt, hopt=hopt, sig2D=sig2D, I0=I0)
         drop(result)
        };
############ End of the function AutoDeSeaTSD(y,s,p,kn,bStart) ###########
