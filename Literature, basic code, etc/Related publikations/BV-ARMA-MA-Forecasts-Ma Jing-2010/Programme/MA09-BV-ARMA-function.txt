############## Begin of the function DeSeaTS(y,s,p,kn,h) ##############
DeSeaTS=function(y,s,p,kn,h)
{
######################### The inputs  #################################
## y - the input seasonal time series data
## s - the seasonal period
## p - 1 or 3, degree of polynomial for estimating the trend function.
## kn- 1, 2, 3 or 4, number of the kernel used as weighting function
         ## kn=1: Uniform Kernel       kn=2: Epanecknikov Kernel
         ## kn=3: Bisquare Kernel      kn=4: Triweight Kernel
## h - the bandwidth
         ## Tip 1: h should be smaller than n/2
         ## Tip 2: Try a bandwidth between 1.5*s and 6*s
 
######################## The basic ideas  ###########################
#### The model: yi = gi + Si +ei                                 ####
#### The approach used here is LWR (locally weighted regression) ####
#### LWR is a linear smoother. The main tesk is to calculate the ####
#### weighting systems for estimating g and S at each point t.   ####
#### In the following only the weighting systems at the left     ####
#### boundary (i=1 to h) and that at i=h+1 will be computed.     ####
#### The weighting system at a point h+1=<i<=n-h is the same.    ####
#### The weightings at i>n-hh are symmetric to that at j=n-i+1.  #### 
#### See Feng (1999), Chapter 7.6 + Feng & Heiler (2000), 786-8. ####

################## constants, vectors and matrices ##################
n=length(y)           ## number of observations
ye=matrix(0,2,n)      ## definition of the matrix of estimates
ht=2*h+1              ## total bandwith

xg=matrix(0,p+s,ht)   ## data matrix for the trend
xs=xg                 ## data matrix for the season
xw=xg                 ## weighted data matrix


  for(j in (1:(p+1))){xg[j,1:ht]=(((1:ht)-h-1)/h)**(j-1)}
  for(j in (1:(trunc((s-1)/2)))){
	xs[p+2*j,  1:ht]=cos(2*j*pi*(1:ht)/s)
	xs[p+2*j+1,1:ht]=sin(2*j*pi*(1:ht)/s)
	}

  if(s/2==trunc(s/2)){
	xs[p+s,1:ht]=cos(pi*(1:ht))
	}

xt=rbind(xg[1:(p+1),], xs[(p+2):(p+s),])



hr=c(2*h-(0:h)) #pointwise right bandwith

     for(i in (1:(n-h-1))){
	     if(i<=h+1){
                   wk=(1-(((1:ht)-i)/(hr[i]+0.5))**2)**(kn-1)

                   for(j in (1:(p+s))){xw[j,]=xt[j,]*wk}



                   yw=solve(xw%*%t(xt))%*%xw
       
                   xgt=t(xg[,i])
                   xst=t(xs[,i])

                   gw=xgt%*%yw
                   ye[1,i]=gw%*%cbind(y[1:ht])
                   sw=xst%*%yw
                   ye[2,i]=sw%*%cbind(y[1:ht])

                   ye[1,n-i+1]=gw%*%cbind(y[n:(n-ht+1)])
                   ye[2,n-i+1]=sw%*%cbind(y[n:(n-ht+1)])

                  }

	     if(i>h+1 && i<n-h){
                           ye[1,i]=gw%*%cbind(y[(i-h):(i+h)])
                           ye[2,i]=sw%*%cbind(y[(i-h):(i+h)])
                          }
        }

#### final results
     result=ye

drop(result)
};
################ End of the function DeSeaTS(y,s,p,kn,h) ################


 
########### Begin of the function DerivativeG(y,s,v,pv,kn,hv) ###########
DerivativeG=function(y,s,v,pv,kn,hv)
{
######################### The inputs  #################################
## v, the order of the derivative
## y, s, pv, kn, hv are as before.

#### Here pv should be larger than v with pv-v odd.               ####
#### See Feng (1999), Chapter 4 + Heiler & Feng (1996, 2000).     ####

################## constants, vectors and matrices ##################
h=hv
p=pv

n=length(y)           ## number of observations
gv=matrix(0,1,n)      ## definition of estimates vector
ht=2*h+1              ## total bandwith

xg=matrix(0,p+s,ht)   ## data matrix for the trend
xs=xg                 ## data matrix for the season
xv=xg                 ## data matrix for the trend
xw=xg                 ## weighted data matrix



        th=(1:ht)-h-1
        for(j in (1:(p+1)))
            {
 	          xg[j,1:ht]=(th/n)**(j-1)
             if(j>v)
                  {
	                Cj=factorial(j)/factorial(j-v)
     	             xv[j,1:ht]=Cj*(th/n)**(j-1-v)
                  }
	         }

        for(j in (1:(trunc((s-1)/2)))){
	       xs[p+2*j,  1:ht]=cos(2*j*pi*th/s)
	       xs[p+2*j+1,1:ht]=sin(2*j*pi*th/s)
	       }

        if(s/2==trunc(s/2)){
       	xs[p+s,1:ht]=cos(pi*th)
	       }

        xt=rbind(xg[1:(p+1),], xs[(p+2):(p+s),])


hr=c(2*h-(0:h)) #pointwise right bandwith

     for(i in (1:(n-h-1))){
	     if(i<=h+1){
                wk=(1-(((1:ht)-i)/(hr[i]+0.5))**2)**(kn-1)

                for(j in (1:(p+s))){xw[j,]=xt[j,]*wk}

                   yw=solve(xw%*%t(xt))%*%xw
                   xvt=t(xv[,i])
       
                   vw=xvt%*%yw
                   gv[i]=vw%*%cbind(y[1:ht])
                   gv[n-i+1]=vw%*%cbind(y[n:(n-ht+1)])*(-1)**v
                  }

	     if(i>h+1 && i<n-h){
                           gv[i]=vw%*%cbind(y[(i-h):(i+h)])
                          }
        }

     gvt=gv          #### Derivative w.r.t. t=i/n
     gvi=gv*(n)**(v) #### Derivative w.r.t. i

#### final results
     result=list(gvt=gvt, gvi=gvi)

drop(result)
};
############ End of the function DerivativeG(y,s,v,pv,kn,hv) ###########



############ Begin of the function VarSeaTSD(y,s) ###########
VarSeaTSD=function(y,s)
{
##################### The inputs  ############################
## y, s are as before.

#### Here two difference-based variance estimates are calculated. ####
#### See Feng (1999), Chapter 7 + Heiler & Feng (1996, 2000).     ####

n=length(y)
n1=s+3
n2=2*s+3

D1=(1:n1)*0
D2=(1:n2)*0

D1[1:3]=c(-1, 2, -1)
D1[(s+1):n1]=c(1, -2, 1)



D2[1:3]=c(-1, 2, -1)
D2[(s+1):(s+3)]=c(2, -4, 2)
D2[(2*s+1):n2]=c(-1, 2, -1)

D1=D1/sqrt(sum(D1**2))
D2=D2/sqrt(sum(D2**2))

sig2D1=0
sig2D2=0

for(i in 1:(n-n1+1)){sig2D1=sig2D1+(D1%*%cbind(y[i:(i+n1-1)]))**2}
for(i in 1:(n-n2+1)){sig2D2=sig2D2+(D2%*%cbind(y[i:(i+n2-1)]))**2}
	
sig2D1=sig2D1/(n-n1+1)
sig2D2=sig2D2/(n-n2+1)

#### final results
     result=list(sig2D1=sig2D1, sig2D2=sig2D2)

drop(result)
};
############ End of the function  VarSeaTSD(y,s) ###########


############ Begin of the function AutoDeSeaTS(y,s,p,kn,bStart) ###########
AutoDeSeaTS=function(y,s,p,kn,bStart, mc)
       {
        ## y, s, p, kn are as before

        ##### input parameters
           bopt=bStart
           hvold=0
           n=length(y)
           k=p+1
           bmax=0.5-1/n     #### The maximal bandwidth 
           bmin=1.5*s/n       #### The minimal bandwidth 

           v=k
           pv=k+1
           runc=1
############# Delta >=0 
           if(k==2){d=0.05}   ## A positive number to
           if(k==4){d=0.10}   ## avoid the boundary effect
           ## 0<= d <=0.15, parameter for the length of the boundary
           n1=trunc(n*d)
           cat("n1=Delta*n=", n1, fill=T)
 

        ###### Kernel constants
           wk=0
           m=100000
           u=(-m:m)/(m+0.5)
           wk=(1-u^2)^(kn-1)
           wk=wk/sum(wk)*m
           R2=sum(wk^2)/(m-0.5)
   
           if(k==2){wkp=wk}
           if(k==4 && kn==2){
	          wkp=15/32*(3-10*u^2+7*u^4)
             wkp=wkp/sum(wkp)*m
            }
           if(k==4 && kn==3){
	          wkp=105/64*(1-5*u^2+7*u^4-3*u**6)
             wkp=wkp/sum(wkp)*m
            }
	  
           Rp=sum(wkp^2)/(m-0.5)
           mukp=sum((u^4)*wkp)/m

        #### two constant in the bandwidth
           c1=factorial(k)**2/(2*k)
           if(mc=="m"){c2=(1-2*d)*(Rp+(s-1)*R2)/(mukp)**2}
           if(mc=="g"){c2=(1-2*d)*Rp/(mukp)**2}

        ####  difference-based estimation of the variance
                sigDC=2   #### 1 or 2
                if(sigDC==1){sig2D=VarSeaTSD(y,s)$sig2D1}
                if(sigDC==2){sig2D=VarSeaTSD(y,s)$sig2D2}
                sig2=sig2D

#####################kmax
            kmax=trunc(n**(1/5))+1
            cat("kmax=", kmax, fill=T)

        #### the main iteration
           noi=25          ### maximal nuber of iterations

           for(i in 1:noi)
             {
              if(runc==1)
                { 	
                cat("This is the ", i, "th iteration.", fill=T)
         	       if(i>=2){hvold=hv}

########## Estimate the c=f from autocovariances
         #### smooth with the selected bandwidth
            hopt=trunc(bopt*n+0.5)
            ye=DeSeaTS(y, s, p, kn, hopt)
            ge=ye[1,]
            se=ye[2,]

            res=y-ge-se
            gammae=avf(res, kmax)
            if(i<5){cf=sig2}
            if(i>=5){cf=(gammae[1]+2*sum(gammae[2:(kmax+1)]))}

             cat("cf=", cf, fill=T)

             cat("gammae", gammae, fill=T)


############# The inflation method    ###############	
	             if(k==2){bv=1*bopt^(5/7)}   #### the most important idea
	             if(k==4){bv=bopt^(9/13)}  #### the most important idea

	              bv=max(bv, bmin)
                 bv=min(bv, bmax)
                 hv=trunc(bv*n+0.5)
                 bvP=trunc(bv*1000+0.5)/1000
                 cat("hv=", hv, "    bv=", bvP)

     
                 gv=DerivativeG(y, s, v, pv, kn, hv)$gvt
                 I2=sum(gv[n1:(n-n1)]**2)/n
                 

                 ### the optimal bandwidth
                    c3=cf/I2 

                    bopt=(c1*c2*c3)**(1/(2*k+1))*n**(-1/(2*k+1))
                    bopt=max(bopt, bmin)
                    bopt=min(bopt, bmax)

                 I0=i

                 hopt=trunc(bopt*n+0.5)
                 boP=trunc(bopt*1000+0.5)/1000              
                 cat("    hopt=", hopt, "    bopt=", boP, fill=T)

                 dh=hvold-hv
                 if(i>=8 && dh==0){runc=0}
                }
             }

         #### smooth with the selected bandwidth
            hopt=trunc(bopt*n+0.5)
            ye=DeSeaTS(y, s, p, kn, hopt)
#            ge=ye[1,]
#
#            hopts=sqrt(s)*hopt
#            if(hopts>(trunc(n/2)-1)){hopts=trunc(n/2)-1}
#            ye=DeSeaTS(y, s, p, kn, hopt)
#            ye[1,]=ge

         #### final results
           result=list(ye=ye, bopt=bopt, hopt=hopt, sig2D=sig2D, sig2S=cf, res=res, I0=I0)
         drop(result)
        };
############ End of the function AutoDeSeaTSD(y,s,p,kn,bStart) ###########

avf=function(x, kmax)
         {avf=0:kmax
          n=length(x)
          for(k in 0:kmax){
	                        avf[k+1]=sum((x[1:(n-k)]-mean(x))*(x[(k+1):n]-mean(x)))/n
                          }                 
          drop(avf)
         }
